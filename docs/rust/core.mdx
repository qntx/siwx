# Core

```rust
use siwx::{Expect, Message, MessageFields, Namespace, authenticate};

let msg = Message::new(MessageFields {
    scheme: None,
    domain: "service.org".into(),
    address: "0xf39Fd6e51aad88F6F4ce6aB8827279cffFb92266".into(),
    statement: None,
    uri: "https://service.org/login".into(),
    version: "1".into(),
    chain_id: "1".into(),
    nonce: siwx::nonce::generate(),
    issued_at: "2026-01-01T00:00:00.000Z".into(),
    expiration_time: None,
    not_before: None,
    request_id: None,
    resources: None,
})?;
let expect = Expect::new("service.org", &nonce)
    .with_chain_id("1")
    .with_max_age(time::Duration::minutes(5));
let auth = authenticate(&evm, text.as_bytes(), &sig, &expect).await?;
```

`expect.check(&message)` runs domain → nonce → scheme → uri → chainId →
requestId (`Mismatch`), then `Expired`/`NotYetValid`/`IssuedAtTooOld`.
